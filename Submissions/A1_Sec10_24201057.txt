public static Integer rowRotation(Integer examWeek, String[][] matrix) {

        Integer rowNum = 0;

        for (int i = 0; i < matrix.length; i++) {

            int row = i;
            if (i - (examWeek - 1) >= 0) {
                row = i - (examWeek - 1);
            } else {
                row = i - (examWeek - 1) + (matrix.length);
            }

            for (int j = 0; j < matrix[0].length; j++) {
                System.out.print("| " + matrix[row][j] + " ");
                if (matrix[row][j] == "AA") {
                    rowNum = row - examWeek;
                }
            }

            System.out.println(" | ");

        }

        return rowNum;
    }

public static Integer[][] compressMatrix(Integer[][] matrix) {

        Integer[][] returnedArray = new Integer[matrix.length / 2][matrix[0].length / 2];

        for (int i = 0; i < matrix.length; i += 2) {
            for (int j = 0; j < matrix[0].length; j += 2) {

                returnedArray[i / 2][j / 2] = matrix[i][j]
                        + matrix[i + 1][j] + matrix[i][j + 1] + matrix[i + 1][j + 1];

            }
        }

        return returnedArray;
    }

public static void playGame(Integer[][] arena) {

        int pointCount = 0;

        for (int i = 0; i < arena.length; i++) {
            for (int j = 0; j < arena[0].length; j++) {

                if (arena[i][j] > 2) {
                    if (i > 0) {
                        if (arena[i - 1][j] == 2) {
                            pointCount += 2;
                        }
                    }
                    if (i < arena.length - 1) {
                        if (arena[i + 1][j] == 2) {
                            pointCount += 2;
                        }
                    }
                    if (j > 0) {
                        if (arena[i][j - 1] == 2) {
                            pointCount += 2;
                        }
                    }
                    if (j < arena.length - 1) {
                        if (arena[i][j + 1] == 2) {
                            pointCount += 2;
                        }
                    }
                }
            }
        }

        if (pointCount >= 10) {
            System.out.println("Points Gained: " + pointCount + ". Your team has survived the game.");
        } else {
            System.out.println("Points Gained: " + pointCount + ". Your team is out.");
        }

    }

    public static void rotation(Character[][] board, int index) {

        char tempValue = board[index][index];
        for (int i = index; i < board.length - 1 - index; i++) {
            board[i][index] = board[i + 1][index];
        }
        for (int j = index; j < board.length - 1 - index; j++) {
            board[board.length - 1 - index][j] = board[board.length - 1 - index][j + 1];
        }
        for (int i = board.length - 1 - index; i > index; i--) {
            board[i][board.length - 1 - index] = board[i - 1][board.length - 1 - index];
        }
        for (int j = board.length - 1 - index; j > index; j--) {
            board[index][j] = board[index][j - 1];
        }
        board[index][index + 1] = tempValue;
    }
